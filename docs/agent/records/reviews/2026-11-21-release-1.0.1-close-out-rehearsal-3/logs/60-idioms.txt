=== the step-5 trap structure, isolated (idiom/traps.sh)
--- case guarded-sub-pass
passed: x=content
done
end
exit=0
--- case guarded-sub-fail
fatal: git-like error
close-out: no FILE at main
exit=1
--- case unguarded-sub-fail
exit=128
--- case step5-unguarded
fatal: a git command failed
close-out: stopped in step 5; the comment at the top of SELF says how to recover
exit=128
--- case step5-fail
close-out: master's first parent is not 1.0.0
close-out: stopped in step 5; the comment at the top of SELF says how to recover
exit=1
--- case step5-sub-in-test
close-out: a comparison
close-out: stopped in step 5; the comment at the top of SELF says how to recover
exit=1
--- case success
done
end
exit=0
--- case after-success-fail
done
exit=3
--- case step5-sleep, SIGINT to the process (job control on)

exit=130

exit=130
